    <script>
      const marketDateElement = document.getElementById('marketDate');
      const ratesTable = document.getElementById('ratesTable');
      const fromCurrency = document.getElementById('fromCurrency');
      const toCurrency = document.getElementById('toCurrency');
      const resultText = document.getElementById('resultText');

      async function loadMarketData() {
        const response = await fetch('/api/rates');
        const data = await response.json();

        marketDateElement.textContent = `Market date: ${data.date} | Base: ${data.base_currency}`;

        const options = Object.keys(data.rates || {}).sort();
        const selectedFrom = 'USD';
        const selectedTo = 'EUR';

        fromCurrency.innerHTML = options.map(code => `<option value="${code}">${code}</option>`).join('');
        toCurrency.innerHTML = options.map(code => `<option value="${code}">${code}</option>`).join('');

        fromCurrency.value = selectedFrom;
        toCurrency.value = selectedTo;

        ratesTable.innerHTML = options.map(code => {
          const rate = Number(data.rates[code]).toFixed(4);
          return `<tr><td>${code}</td><td>${rate}</td></tr>`;
        }).join('');

        await convertCurrency();
      }

      async function convertCurrency() {
        const amount = Number(document.getElementById('amount').value || 0);
        const from = fromCurrency.value;
        const to = toCurrency.value;

        const response = await fetch('/api/convert', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ amount, from_currency: from, to_currency: to })
        });

        const data = await response.json();
        if (!response.ok) {
          resultText.textContent = data.error || 'Conversion failed.';
          return;
        }

        resultText.textContent = `${Number(data.amount).toFixed(2)} ${data.from_currency} = ${Number(data.converted_amount).toFixed(4)} ${data.to_currency} (rate: ${Number(data.rate).toFixed(4)}, date: ${data.date})`;
      }

      document.getElementById('convertBtn').addEventListener('click', convertCurrency);
      document.getElementById('amount').addEventListener('input', convertCurrency);
      fromCurrency.addEventListener('change', convertCurrency);
      toCurrency.addEventListener('change', convertCurrency);

      loadMarketData();
    </script>
